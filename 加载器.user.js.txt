// ==UserScript==
// @name         加载器
// @namespace    https://github.com/HuchiLiunai/my-js
// @version      1.0.0
// @match        https://test.baidu.com/argus/plaza/project/1/verify/baseQueueId/*
// @match        https://test.baidu.com/argus/plaza/project/mt/verify/baseQueueId/*
// @grant        GM_xmlhttpRequest
// @grant        GM_setClipboard
// @grant        unsafeWindow
// @connect      test.baidu.com
// @connect      cdn.jsdelivr.net
// @run-at       document-start
// ==/UserScript==

(function () {
    'use strict';

    const REPO = 'HuchiLiunai/my-js';
    const XOR_KEY = 'FeFMYP24ajqQPF9OMi2OeXda9wq4o4qC';
    const CONFIG_URL = `https://cdn.jsdelivr.net/gh/${REPO}@latest/config.enc.js`;

    const win = typeof unsafeWindow !== 'undefined' ? unsafeWindow : window;

    (function earlyHook() {
        const RE_GET = /\/result\/verify\/project\/[^/]+\/base-queue\/[^/]+\/get(\?|$)/;
        const RE_LIST = /\/result\/verify\/queue\/[^/]+\/list(\?|$)/;
        const RE_REFS = /\/video\/project\/[^/]+\/base-queue\/[^/]+\/page\/(\d+)\/references/;
        const MAX = 300;

        const Q = win.__myQueue = win.__myQueue || { get: [], list: [], refs: [] };
        const push = (arr, item) => {
            if (!item) return;
            arr.push(item);
            while (arr.length > MAX) arr.shift();
        };

        if (win.fetch && !win.fetch.__qHooked) {
            const orig = win.fetch;
            win.fetch = function (input) {
                const url = typeof input === 'string' ? input : (input && input.url) || '';
                const p = orig.apply(this, arguments);
                if (RE_GET.test(url) || RE_LIST.test(url) || RE_REFS.test(url)) {
                    p.then(r => r.clone().json()).then(json => {
                        if (RE_LIST.test(url)) push(Q.list, json);
                        else if (RE_GET.test(url)) push(Q.get, json);
                        else push(Q.refs, { url, json });
                    }).catch(() => {});
                }
                return p;
            };
            win.fetch.__qHooked = true;
        }

        const XHR = win.XMLHttpRequest;
        if (XHR && XHR.prototype && !XHR.prototype.__qHooked) {
            const origOpen = XHR.prototype.open;
            const origSend = XHR.prototype.send;
            XHR.prototype.open = function (m, u) {
                this.__qUrl = u;
                return origOpen.apply(this, arguments);
            };
            XHR.prototype.send = function () {
                const url = String(this.__qUrl || '');
                if (RE_GET.test(url) || RE_LIST.test(url) || RE_REFS.test(url)) {
                    this.addEventListener('loadend', function () {
                        let json = null;
                        try { json = JSON.parse(this.responseText); } catch (e) {}
                        if (!json) return;
                        if (RE_LIST.test(url)) push(Q.list, json);
                        else if (RE_GET.test(url)) push(Q.get, json);
                        else push(Q.refs, { url, json });
                    });
                }
                return origSend.apply(this, arguments);
            };
            XHR.prototype.__qHooked = true;
        }
    })();

    function gmFetchText(url) {
        return new Promise(resolve => {
            GM_xmlhttpRequest({
                method: 'GET',
                url,
                timeout: 8000,
                onload: r => resolve(r.status === 200 ? r.responseText : null),
                onerror: () => resolve(null),
                ontimeout: () => resolve(null),
            });
        });
    }
    const gmFetchJson = url => gmFetchText(url).then(t => { try { return JSON.parse(t); } catch (e) { return null; } });

    function xorDecode(b64, key) {
        let bin;
        try { bin = atob(b64); } catch (e) { return null; }
        let out = '';
        for (let i = 0; i < bin.length; i++) {
            out += String.fromCharCode(bin.charCodeAt(i) ^ key.charCodeAt(i % key.length));
        }
        try { return decodeURIComponent(escape(out)); } catch (e) { return out; }
    }

    function parseConfig(text) {
        if (!text) return null;
        const m = text.match(/"([A-Za-z0-9+/=]+)"/);
        const b64 = m ? m[1] : text.trim();
        const json = xorDecode(b64, XOR_KEY);
        if (!json) return null;
        try { return JSON.parse(json); } catch (e) { return null; }
    }

    (async () => {
        try {
            const userJson = await gmFetchJson('https://test.baidu.com/argus/api/user/current_user');
            if (!userJson || userJson.errorCode !== 0) return;
            const uid = String((userJson.data && userJson.data.id) || '');
            if (!uid) return;

            const cfgText = await gmFetchText(CONFIG_URL + '?t=' + Date.now());
            const config = parseConfig(cfgText);
            if (!config) return;

            if (config.enabled === false) return;

            if (!Array.isArray(config.whitelist)) return;
            if (!config.whitelist.map(String).includes(uid)) return;

            const tw = config.timeWindow || {};
            const hour = new Date().getHours();
            if (typeof tw.startHour === 'number' && hour < tw.startHour) return;
            if (typeof tw.endHour === 'number' && hour >= tw.endHour) return;

            if (!config.coreUrl) return;
            const coreText = await gmFetchText(config.coreUrl + '?t=' + Date.now());
            if (!coreText || coreText.trim().length < 20) return;

            win.__myEnv = {
                unsafeWindow: win,
                GM_setClipboard: typeof GM_setClipboard !== 'undefined' ? GM_setClipboard : null,
                UID: uid,
                CONFIG: config,
            };
            try { new Function(coreText)(); } catch (e) {}
        } catch (e) {}
    })();
})();
